// timescale.cfg
// Steps host_timescale through 0.5 / 0.75 / 1 / 1.5 / 2 / 10
// mouse5 (forward side button) = speed up, mouse4 (back side button) = slow down
// Requires sv_cheats 1 (practice / local server only)
// Run with: exec timescale

alias ts_200 "host_timescale 2;    alias ts_up ts_max; alias ts_down ts_150; echo [timescale] 2x"
alias ts_max "host_timescale 10;   alias ts_up ts_max; alias ts_down ts_200; echo [timescale] 10x  (MAX)"

// start at normal speed
ts_100

bind mouse4 ts_down
bind mouse5 ts_up

// --- bot toughness: T-side takes 0.2x damage = effectively 500 HP ---
// AK headshot vs helmet: 111 x 0.2 = 22.2 -> 4 headshots = 88.8 (alive), 5th kills

mp_damage_scale_ct_head 0.2

// --- bullet impact markers ---

alias impmode_3 "sv_showimpacts 3; alias impmode_next impmode_0; echo [impacts] mode: server only (blue)"

impmode_1
bind p impmode_next

// O toggles wallbang/penetration info text at impact points (damage, surface thickness)
sv_showimpacts_penetration 1

// +/- keys (main row and numpad) step how long impacts stay before fading

alias imp_2   "sv_showimpacts_time 2;   alias imp_up imp_4;   alias imp_down imp_1;   echo [impacts] fade 2s"
